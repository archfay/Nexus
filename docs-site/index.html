Быстрый старт

Создайте файл mymodule.py в папке loaded_modules/:

from .. import loader, utils

class MyModuleMod(loader.Module):
    """Описание вашего модуля"""
    
    strings = {
        "name": "MyModule",
        "hello": "👋 Привет, {}!"
    }
    
    @loader.command()
    async def hello(self, message):
        """Приветствие пользователя"""
        await utils.answer(message, self.strings["hello"].format(
            message.sender.first_name
        ))

Загрузите модуль командой: .loadmod mymodule

Команда с reply

@loader.command()
async def info(self, message):
    """Информация о сообщении"""
    reply = await message.get_reply_message()
    if not reply:
        await utils.answer(message, "❌ Ответьте на сообщение!")
        return
    
    info = f"👤 От: {reply.sender_id}\n📅 Дата: {reply.date}"
    await utils.answer(message, info)

from .. import loader, utils

class MessageCounterMod(loader.Module):
    """Счетчик сообщений в чате"""

    @loader.watcher()
    async def watcher(self, message):

Напоминания

from .. import loader, utils
import asyncio

class ReminderMod(loader.Module):
    """Модуль напоминаний"""
    
    strings = {"name": "Reminder"}
    
    @loader.command()
    async def remind(self, message):
        """<секунды> <текст> - Напоминание"""
        args = utils.get_args_raw(message)
        try:
            seconds, text = args.split(maxsplit=1)
            seconds = int(seconds)
        except:
            await utils.answer(message, "❌ Использование: .remind <сек> <текст>")
            return
        
        await utils.answer(message, f"⏰ Напомню через {seconds} сек")
        await asyncio.sleep(seconds)
        await message.reply(f"🔔 Напоминание: {text}")

Калькулятор

from .. import loader, utils

class CalculatorMod(loader.Module):
    """Простой калькулятор"""
    
    strings = {"name": "Calculator"}
    
    @loader.command()
    async def calc(self, message):
        """<выражение> - Вычислить"""
        expr = utils.get_args_raw(message)
        if not expr:
            await utils.answer(message, "❌ Укажите выражение!")
            return
        
        try:
            result = eval(expr, {"__builtins__": {}})
            await utils.answer(message, f"🔢 {expr} = {result}")
        except Exception as e:
            await utils.answer(message, f"❌ Ошибка: {e}")

Безопасность

⚠️ Важные правила безопасности

  • Никогда не используйте eval() с пользовательским вводом без ограничений
  • Не храните токены и пароли в открытом виде
  • Проверяйте права доступа перед выполнением критичных операций
  • Используйте try-except для обработки ошибок
  • Не доверяйте данным от пользователей

Проверка прав

@loader.command()
async def admin_cmd(self, message):
    """Команда только для владельца"""
    me = await self.client.get_me()
    if message.sender_id != me.id:
        return  # Игнорируем чужие команды
    
    # Выполняем команду
    await utils.answer(message, "✅ Выполнено")

Безопасная работа с API

@loader.command()
async def apikey(self, message):
    """<ключ> - Сохранить API ключ"""
    key = utils.get_args_raw(message)
    if not key:
        await utils.answer(message, "❌ Укажите ключ!")
        return
    
    # Сохраняем в БД (не в коде!)
    self.db.set("MyModule", "api_key", key)
    
    # Удаляем сообщение с ключом
    await message.delete()
    
    await self.client.send_message(
        message.chat_id,
        "✅ API ключ сохранен"
    )

Часто задаваемые вопросы

Как загрузить модуль?

Используйте команду .loadmod имя_файла или .dlmod URL для загрузки из интернета.

Где хранятся модули?

Модули хранятся в папке loaded_modules/ в корне проекта.

Как обновить модуль?

Используйте .unloadmod имя затем .loadmod файл или просто перезагрузите бота.

Можно ли использовать внешние библиотеки?

Да, установите их через pip install и импортируйте в модуле.

Как сделать команду приватной?

Проверяйте message.sender_id и сравнивайте с вашим ID.
